=== dbdiagram.io version (cleaner alternative) ===

Table ward { ward_id int [pk]  municipality varchar  ward_no int  address varchar  opens_at time  closes_at time }
Table service { service_id int [pk]  ward_id int [ref: > ward.ward_id]  name_en varchar  name_ne varchar  fee_npr decimal  slot_minutes int  room varchar  needs_signature boolean  active boolean }
Table officer_availability { ward_id int [ref: > ward.ward_id]  unavailable_on date  reason varchar  set_by_staff_id int [ref: > staff.staff_id]  set_at timestamp  indexes { (ward_id, unavailable_on) [pk] } }
Table required_document { document_id int [pk]  service_id int [ref: > service.service_id]  description varchar  mandatory boolean }
Table slot { slot_id int [pk]  service_id int [ref: > service.service_id]  slot_date date  start_time time  open boolean }
Table citizen { citizen_id int [pk]  full_name varchar  mobile_no varchar [unique]  created_at timestamp }
Table appointment { appointment_id int [pk]  reference_no varchar [unique]  slot_id int [unique, null, ref: - slot.slot_id]  service_id int [ref: > service.service_id]  visit_date date  citizen_id int [ref: > citizen.citizen_id]  status varchar  walk_in boolean  created_at timestamp }
Table staff { staff_id int [pk]  ward_id int [ref: > ward.ward_id]  username varchar [unique]  password_hash varchar  role varchar }
Table audit_log { audit_id int [pk]  appointment_id int [ref: > appointment.appointment_id]  staff_id int [ref: > staff.staff_id]  old_status varchar  new_status varchar  changed_at timestamp }
